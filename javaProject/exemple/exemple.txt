typedef int Entier;
typedef <Entier, Entier> PaireEntier;

void main() {

	//Opération et déclaration de variable
	int i = 1;
	int j = 2;
	i = j + i;
	
	print i; //3
	print j; //2
	
	//Couple
	<int, int> p = <3, 4>;
	print (fst p); //3
	print (snd p); //4
	
	//Pointeur
	int v = 5;
	int *ptr = &v;
	print (*ptr); //5
	*ptr = 6;
	print v; //6
	
	//Tableau + WHILE
	int tab [] = new int[5];
	int k = 0;
	while (k < 5) {
		k = k+1;
		tab[k-1] = k;
	}
	k = 0;
	while (k<5) {
		print (tab[k]); //12345
		k = k+1;
	}
	
	//CONSTANTE et IF ELSE
	const int LIMITAGE = 18;
    	int age = 18;
    	boolean verif = true;
    	
    	if (age < LIMITAGE) {
		print verif;
	    	verif = true;
	} else {
	    	verif = false;
	}    
	print verif; // 0
	print age; // 18
	
	//IF
	int age2 = 10;
	if (age2 < LIMITAGE) {
		print age2; // 10
	}
}
