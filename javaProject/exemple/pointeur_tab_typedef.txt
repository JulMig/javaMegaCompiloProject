typedef int Entier;
typedef <Entier, Entier> PaireEntier;

void main() {

    Entier i = 1;
    PaireEntier p = <i, 2>; //<1, 2>
    PaireEntier a [] = new PaireEntier[3];
    
    a[0] = p; //<1, 2>
    a[1] = <3, 4>; //<3, 4>
    
    print (fst (a[0]));  //1
    print (snd (a[0]));  //2
    print (fst (a[1]));  //3
    print (snd (a[1]));  //4

    
    PaireEntier v = <8, 9>; //<8, 9>
    PaireEntier *ptr = &v; //ptr -> <8,9>
    print (fst v); //8
    print (snd (*ptr)); //9
    *ptr = <2, 3>; // v -> <2, 3>
    print (snd v); //3
    print (fst (*ptr)); //2
    
    //12348932
    
    
}
