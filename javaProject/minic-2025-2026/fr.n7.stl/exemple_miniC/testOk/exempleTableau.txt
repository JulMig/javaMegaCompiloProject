typedef int Entier;
typedef <Entier, Entier> PaireEntier;

//Exemple tableau avec type défini
void main() {

	Entier i = 1;
    	PaireEntier p = <i, 2>; //<1, 2>
    	PaireEntier a [] = new PaireEntier[2];
    
    	a[0] = p; //<1, 2>
    	a[1] = <3, 4>; //<3, 4>
    
    	print (fst (a[0]));  //1
    	print (snd (a[0]));  //2
    	print (fst (a[1]));  //3
    	print (snd (a[1]));  //4
	

}
